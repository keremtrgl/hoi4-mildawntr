scripted_gui = {

	# Shown when +Turkey's TUR country history did not run: another mod overrides history/countries/TUR - Turkey.txt
	# (typically a second Turkey mod loaded after us), so Turkey starts without our parties, leaders and spirits.
	# Detection: our TUR history sets TUR_submod_history_loaded; on_startup sets TUR_load_order_wrong if it is missing.
	# (Millennium Dawn itself always loads first: the game orders a mod after its dependencies, tested 2026-10-28.)
	TUR_load_order_warning = {
		context_type = player_context

		window_name = "TUR_load_order_warning_window"

		visible = {
			is_ai = no
			has_global_flag = TUR_load_order_wrong
			NOT = { has_country_flag = TUR_load_order_warning_closed }
		}

		effects = {
			TUR_load_order_close_button_click = {
				set_country_flag = TUR_load_order_warning_closed
			}
		}
	}
}
