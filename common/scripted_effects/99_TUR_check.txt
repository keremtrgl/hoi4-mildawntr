# Clears the sectarian state modifiers once Turkey's internal divisions are resolved,
# flagging each state by the group that used to dominate it.
TUR_remove_state_modifier_effects = {
	set_global_flag = BLOCK_SECTARIAN_ADDITIONS
	hidden_effect = {
		# Kurdish regions
		every_owned_state = {
			limit = { has_dynamic_modifier = { modifier = conscription_exemptions_granted } }
			remove_dynamic_modifier = { modifier = conscription_exemptions_granted }
			set_state_flag = TUR_kurdish_state_conquered
		}
		# Islamist / traditionalist regions
		every_owned_state = {
			limit = {
				OR = {
					has_dynamic_modifier = { modifier = islamist_opposition }
					has_dynamic_modifier = { modifier = devout_islamists }
					has_dynamic_modifier = { modifier = islamists_placated }
				}
			}
			remove_dynamic_modifier = { modifier = islamist_opposition }
			remove_dynamic_modifier = { modifier = devout_islamists }
			remove_dynamic_modifier = { modifier = islamists_placated }
			set_state_flag = TUR_traditionalist_state_conquered
		}
		# Kemalist regions
		every_owned_state = {
			limit = { has_dynamic_modifier = { modifier = kemalist_loyalties } }
			remove_dynamic_modifier = { modifier = kemalist_loyalties }
			set_state_flag = TUR_loyal_kemalist_state_conquered
		}
		every_owned_state = {
			limit = { has_dynamic_modifier = { modifier = kemalist_schemers } }
			remove_dynamic_modifier = { modifier = kemalist_schemers }
		}
	}
}
